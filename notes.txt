#Create a new database

#Create a new user with a password

#Grant the user access to the database:
CREATE ROLE
GRANT ALL PRIVILEGES ON DATABASE social_book_db TO yushi;

-- Set schema ownership explicitly
ALTER SCHEMA public OWNER TO yushi;

-- Explicitly allow the user to create tables and other objects
GRANT ALL ON SCHEMA public TO yushi;
GRANT CREATE ON SCHEMA public TO yushi;

-- Allow the user full permissions for future created tables/sequences
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT ALL PRIVILEGES ON TABLES TO yushi;
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT ALL PRIVILEGES ON SEQUENCES TO yushi;

-- Just to ensure no permission issues persist
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO yushi;
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA public TO yushi;
