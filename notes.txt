-- Explicitly allow the user to create tables and other objects

-- Allow the user full permissions for future created tables/sequences

ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT ALL PRIVILEGES ON SEQUENCES TO yushi;

-- Just to ensure no permission issues persist
